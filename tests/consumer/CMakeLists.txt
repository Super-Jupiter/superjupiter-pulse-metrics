cmake_minimum_required(VERSION 3.16)
project(SuperJupiterPulseMetricsConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_PULSE_METRICS_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_PULSE_METRICS_SOURCE_DIR}" pulse-metrics)
  if(TARGET superjupiter-pulse-metrics-tests OR TARGET superjupiter-pulse-metrics-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterPulseMetrics 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-pulse-metrics-consumer main.cpp)
target_link_libraries(superjupiter-pulse-metrics-consumer PRIVATE SuperJupiter::PulseMetrics)
